5. EXACT NARROW RUNNER SOURCE (saved as ../cf089-probe.py for section 3)

"""Run selected real CF-089 selftest arms without unrelated repo arms."""
import argparse
import subprocess
from pathlib import Path

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument("--arm", type=int)
args = parser.parse_args()
text = Path("tests/selftest.sh").read_text()
block = text.split("# BEGIN census-fail-closed\n", 1)[1].split(
    "# END census-fail-closed", 1
)[0]
if args.arm is not None:
    helpers = block.split('say "22.', 1)[0]
    marker = f'say "{args.arm}.'
    arm = marker + block.split(marker, 1)[1]
    arm = arm.split(f'say "{args.arm + 1}.', 1)[0]
    block = helpers + arm
script = (
    'set -uo pipefail\nHERE="$PWD"\n. "$HERE/scripts/lib.sh"\nfails=0\n'
    + block
    + '\n[ "$fails" -eq 0 ] && { say "CENSUS SELFTEST: PASS"; exit 0; }\n'
    + 'say "CENSUS SELFTEST: FAIL ($fails)"; exit 1\n'
)
result = subprocess.run(["bash"], input=script, text=True, check=False)
raise SystemExit(result.returncode)
